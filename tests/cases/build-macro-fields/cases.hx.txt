--- local-class-has-no-fields-while-it-builds
@:autoBuild(M.show())
class A { public var a:Int = 1; public static var sa = 0; public function new() {} }
class B extends A { public var b:Int = 2; static var sb = 3; public function new() { super(); } }
class P { static function main() { new B(); } }

--- fields-are-checked-after-the-last-build
@:autoBuild(M.dropInherited())
@:autoBuild(M.keep())
class A { public var a:Int = 1; public function new() {} }
class B extends A { public var a:Int = 2; public var b:Int = 3; }
class P { static function main() { new B().b; } }

--- the-class-has-no-fields-between-its-builds

--- a-subclass-has-no-fields-while-it-builds-its-parent
@:build(M.peek(["B"]))
@:autoBuild(M.show())
class A { public var a:Int = 1; public function new() {} }
class B extends A { public var b:Int = 2; }
class P { static function main() { new B().b; } }

--- a-subclass-the-parents-build-reaches-builds-after-it
class B extends A { public var b:Int = 2; }
class C extends A { public var c:Int = 3; }
@:build(M.peek(["B", "C"]))
@:autoBuild(M.show())
class A { public var a:Int = 1; public function new() {} }
class P { static function main() { new B().b + new C().c; } }

--- a-grandchild-waits-on-the-same-parent
class D extends B { public var d:Int = 4; }
class B extends A { public var b:Int = 2; }
@:build(M.peek(["D", "B"]))
@:autoBuild(M.show())
class A { public var a:Int = 1; public function new() {} }
class P { static function main() { new D().d; } }

--- the-macro-api-builds-a-canceled-class-once-its-parent-is-built

@:build(M.peek(["B"]))
class E { public function new() {} }
@:build(M.peek(["B"]))
@:autoBuild(M.show())
