# The server app's Shared.ConfigSettings/DataPoolDef, /AuthenticationDef and Shared.AITools.ToolPlugin/ToolFunction
# are nested types of a --net-lib assembly: their @:native joins the enclosing types with dots, and
# DotnetTypes.externPathOf split it into a namespace. hxc 4b17dec9: six "the .NET type LibA.Outer.Inner
# (.Deep) outside the referenced assemblies" errors. TypeRefs must nest (LibA.Outer/Inner/Mode), as gencs's.
--- netlib-nested-types @gencs @netlib=LibA
import liba.Outer;

class P {
	static function main() {
		var i = Outer.Make(3);
		trace(i.X);
		trace(Outer_Inner.Hello());
		var j = new Outer_Inner(4);
		trace(j.X + i.X);
		trace(j.M == Outer_Inner_Mode.B);
		var m:Outer_Inner_Mode = j.M;
		trace(m);
		var o = new Outer();
		trace(o.Items.Count);
		trace(new Outer_Inner_Deep().D);
		var f:Outer_Inner->Int = function(x:Outer_Inner) return x.X * 10;
		trace(f(j));
	}
}

# A hand-written extern over a framework type that no --net-std/--net-lib assembly defines (the server app's
# CSTools.hx JsonSerializerOptions, License.hx:448/462). No --net-lib, so it runs in today's harness.
# gencs: compiles (dotnet build references the whole net10.0 framework). hxc 4b17dec9: "the .NET type
# System.Text.Json.JsonSerializerOptions outside the referenced assemblies". hxc must find it in the
# .NET 10 reference pack and write AssemblyRef System.Text.Json 10.0.0.0 cc7b13ffcd2ddd51, as gencs's build does;
# IncludeFields/MaxDepth are properties (set_IncludeFields/get_MaxDepth), which only the metadata says.
--- netlib-framework-extern @gencs
// A hand-written extern over a framework type that no --net-std/--net-lib assembly defines, as the server app's
// CSTools.hx (System.Text.Json.JsonSerializerOptions).
@:native("System.Text.Json.JsonSerializerOptions") extern class JsonSerializerOptions {
	public function new();
	public var IncludeFields:Bool;
	public var MaxDepth:Int;
}

class P {
	static function main() {
		var opts = new JsonSerializerOptions();
		opts.IncludeFields = true;
		trace(opts.IncludeFields);
		trace(opts.MaxDepth);
		var o:Dynamic = opts;
		trace(Std.isOfType(o, JsonSerializerOptions));
	}
}

# A type reached through a type forwarder: hxcs's net-20 std has no System.Numerics.BigInteger; LibA

# hxc 4b17dec9: "the .NET type System.Numerics.BigInteger outside the referenced assemblies".
--- netlib-forwarded-extern @gencs @netlib=LibA
// A hand-written extern over a type that no --net-std assembly (hxcs's net-20) defines, reached through
// LibA's reference to netstandard, whose reference-pack facade forwards it to System.Runtime.Numerics.

	function new(v:Int):Void;

# haxeui-cs's shape (Native.hx: `@:native("HaxeUICs.UIPanel") extern class UIPanel extends Control`,
# Control = Avalonia.Controls.Control): hand-written externs whose metadata supertype lives in an
# assembly passed only beside the --net-lib one. ExternMembers.chain walks LibA.Panel's metadata
# super, LibB.BaseCtl. hxc 4b17dec9: "the .NET type LibB.BaseCtl outside the referenced assemblies".
# hxc must read LibB.dll from beside LibA.dll and write AssemblyRef LibB 2.3.0.0 (null token).
# NOT @gencs: gencs's C# fails, since hxcs references only the --net-lib files:
#   P.hx(16,61): error CS0012: The type 'BaseCtl' is defined in an assembly that is not referenced.
#   You must add a reference to assembly 'LibB, Version=2.3.0.0, Culture=neutral, PublicKeyToken=null'.
#   (also CS1061 'Panel' does not contain a definition for 'Kind'/'Visible', CS0400 'LibB' not found)
# hxc doing better here is allowed (the spec's Intent); the expectation is hxc's run with the patch.
--- netlib-dependency-beside @netlib=LibA
// Hand-written externs over --net-lib types, as haxeui-cs's Native.hx (UIPanel extends Avalonia's Control).
@:native("LibB.BaseCtl") extern class Ctl {
	var Visible:Bool;
	function Kind():String;
	function Who():String;
}

@:native("LibA.Panel") extern class Panel extends Ctl {
	function new():Void;
	function Me():String;
}

class P {
	static function main() {
		var p = new Panel();
		trace(p.Me());
		trace(p.Who());
		trace(p.Kind());
		trace(p.Visible);
		var c:Ctl = p;
		trace(c.Who());
	}
}
