			Sys.println(label + ": not found");
			return;
		}
		var ifs = [for (i in t.GetInterfaces()) typeName(i)];
		ifs.sort(Reflect.compare);
		Sys.println(label + ": " + typeName(t) + (t.IsInterface ? " interface" : "") + " implements " + ifs.join(","));
		var attrs = t.GetCustomAttributes(false);
		for (i in 0...attrs.Length) {
			var g = cs.Lib.as(attrs[i], cs.internal.HxObject.GenericInterface);
			if (g != null) Sys.println("  [GenericInterface(" + typeName(g.generic) + ")]");
		}

		var ms = t.GetMethods(flags);
		for (i in 0...ms.Length) {
			var n = ms[i].Name;
			if (StringTools.startsWith(n, "__hx_") && n != "__hx_cast") continue;
			Sys.println("  " + Std.string(ms[i]) + (ms[i].IsStatic ? " static" : ""));

// Casts into another instantiation of a generic interface the class implements. gencs's parent cast
// Th_cast<T_c>() calls FP_cast<T>() with the class's own T, so it returns `this` and the castclass after it
// throws InvalidCastException (types notes §4). hxc emits the correct body: FP_cast<T_c>(), Perm_cast<B_c, A_c>(),
// Partial_cast<B_c>(), Both_cast<A_c>(), and in the std StringMap<T> implements IMap<String,T>
// haxe_ds_StringMap_cast<V_c>(); Fixed<T> implements I1<Int> keeps Fixed_cast<T>() as gencs does.
interface Th<T> {
	function then<R>(f:T->R):Th<R>;
	function value():T;
}
class FP<T> implements Th<T> {
	var v:Null<T>;
	public function new(v:T) this.v = v;
	public function then<R>(f:T->R):Th<R> {
		// thenshim's shape: an Any-typed promise cast to Th<R>.
		return cast impl(f);
	}

}
