# Casts notes §6.3: lossy conversions copy through genericCast in both.
--- cast-array-convert @gencs
// Array<Dynamic> -> Array<X> where an element does not convert losslessly: a copy through genericCast in
// gencs and (by the lossless fallback) in hxc. Prints only the converted values. @gencs.
class P {
	static function toInts(a:Array<Dynamic>):Array<Int> {
		var r:Array<Int> = cast a;
		return r;
	}

	static function toFloats(a:Array<Dynamic>):Array<Float> {
		var r:Array<Float> = cast a;
		return r;
	}

	static function toBools(a:Array<Dynamic>):Array<Bool> {
		var r:Array<Bool> = cast a;
		return r;
	}

	static function toStrs(a:Array<Dynamic>):Array<String> {
		var r:Array<String> = cast a;
		return r;
	}

	static function run(name:String, f:Void->Dynamic) {
		try {
			Sys.println(name + ": " + Std.string(f()));
		} catch (e:Dynamic) {
			var s = Std.string(e);
			if (s.length > 100) s = s.substr(0, 100);
			Sys.println(name + ": EXC " + s.split("\n")[0].split("\r")[0]);
		}
	}

	static function main() {
		run("int<-halves", () -> toInts([0.5, 1.5, 2.5, -0.5, -1.5, 1.4, 1.6]));
		run("int<-null", () -> toInts([null, 1]));
		run("int<-big", () -> toInts([1e10]));
		run("int<-nan", () -> toInts([Math.NaN]));
		run("int<-Int64", () -> toInts([haxe.Int64.make(0, 5)]));
		run("int<-bool", () -> toInts([true, false]));
		run("int<-str", () -> toInts(["7", " 8 "]));
		run("int<-badstr", () -> toInts(["1.5"]));
		run("int<-obj", () -> toInts([{}]));
		run("int<-date", () -> toInts([Date.fromTime(0)]));
		run("float<-null", () -> toFloats([null, 1]));
		run("float<-str", () -> toFloats(["1.5"]));
		run("float<-bool", () -> toFloats([true]));
		run("bool<-int", () -> toBools([1]));
		run("bool<-null", () -> toBools([null, true]));
		run("str<-int", () -> toStrs([1]));
		run("str<-obj", () -> toStrs([{}]));
		run("single", () -> { var r:Array<Single> = haxe.Json.parse("[1.1, null]"); Std.string(r); });
		run("int64", () -> { var r:Array<haxe.Int64> = haxe.Json.parse("[1.6]"); Std.string(r); });
	}
}

--- cast-array-lossless-serial @gencs
// Reading a typed field never rewrites stored data: with 1.5 in an Array<Int> field, or null in an
// Array<Float> one, the read copies and the record re-serializes as it was. @gencs.
typedef TI = {val:Array<Int>};
typedef TF = {val:Array<Float>};

class P {
	static function main() {
		var t:TI = haxe.Json.parse('{"val":[1.5,2,null]}');
		var v = t.val;
		Sys.println(Std.string(v) + " " + haxe.Serializer.run(t));
		var u:TF = haxe.Unserializer.run("oy3:valad1.5i2nhg");
		var w = u.val;
		Sys.println(Std.string(w) + " " + haxe.Serializer.run(u));
		var x:TI = haxe.Json.parse('{"val":[1,2]}');
		var y = x.val;
		Sys.println(Std.string(y) + " " + haxe.Serializer.run(x));
		var z:TF = haxe.Json.parse('{"val":[1,2]}');
		var q = z.val;
		Sys.println(Std.string(q) + " " + haxe.Serializer.run(z));
	}
}

// gencs: writes through a typed read of an unserialized array are lost. Recorded #8314 divergence: hxc keeps them.
typedef DataItem = {data:String, a:Array<String>};

class P {
	static function run(name:String, f:Void->Dynamic) {
		try {
			Sys.println(name + ": " + Std.string(f()));
		} catch (e:Dynamic) {
			var s = Std.string(e);
			if (s.length > 100) s = s.substr(0, 100);
			Sys.println(name + ": EXC " + s.split("\n")[0].split("\r")[0]);
		}
	}

	static function main() {
		run("body", () -> {
			var simpleObj:{val:Array<Float>} = {val: [0, 0, 0, 0]};

			var newVal2:Int = 2;

			var afterSerialize:{val:Array<Float>} = haxe.Unserializer.run(haxe.Serializer.run(simpleObj));
			afterSerialize.val[0] = newVal2;

			afterSerialize.val = afterSerialize.val.copy();

			var a = t.a;
			t.a.push("raw anonymous object is ok");
			var r1 = Std.string(a) + " " + Std.string(t.a);
			var t:DataItem = haxe.Json.parse(haxe.Json.stringify({data: "d", a: []}));
			var a = t.a;
			t.a.push("after json conversion fails");
			r1 + " | " + Std.string(a) + " " + Std.string(t.a) + " same=" + (t.a == t.a);

# Recorded #8314 divergence (types notes §8.17, probe nested; overlaps cast-array-nested through class
# fields). gencs: inner write 1, same inner false, ai same false, ai push seen 1. Expected [run]: types §8.17's
# stated lines (inner write 5, same inner true, ai push seen 2, ac/outer as gencs) and `ai same: true`, which

# cast-4872-eq's `da == db` does.
--- nested-array-element-cast
// Nested arrays under erasure: Array<Array<Int>> is Array<object>, so the inner cast happens at each element read.
class C { public var x = 1; public function new() {} }
class H {
	public var aai:Array<Array<Int>>;
	public var ac:Array<C>;
	public var ai:Array<Int>;
	public function new() {}
}
class P {
	static function main() {
		var s = haxe.Serializer.run({aai: [[1, 2]], ac: [], ai: [1]});
		var o:Dynamic = haxe.Unserializer.run(s);
		var h = new H();
		h.aai = o.aai;
		h.ac = o.ac;
		h.ai = o.ai;
		h.aai[0][0] = 5;
		Sys.println("inner write: " + h.aai[0][0]);
		var inner = h.aai[0];
		Sys.println("same inner: " + (inner == h.aai[0]));
		Sys.println("outer same: " + (h.aai == o.aai));
		Sys.println("ac same: " + (h.ac == o.ac));
		Sys.println("ai same: " + (h.ai == o.ai));
		h.ac.push(new C());
		Sys.println("ac push seen: " + o.ac.length);
		h.ai.push(2);
		Sys.println("ai push seen: " + o.ai.length);
		var t:{ac:Array<C>} = o;
		t.ac.push(new C());
		Sys.println("anon Array<C> push seen: " + o.ac.length);
	}
}
