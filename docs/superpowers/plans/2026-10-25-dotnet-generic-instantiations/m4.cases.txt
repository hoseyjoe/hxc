# hxcs's Array @:classCode indexer: the get_Item/set_Item entries removed from generic-reflection,
# resolve-generic-class and type-argument-clr-shapes, in a program of its own (its member lists follow DCE).
--- array-companion-indexer @gencs
// hxcs's Array has a @:classCode indexer (`T this[int]`, and `object this[int]` on the companion), which shows
// in reflection as get_Item/set_Item. hxc ignores @:classCode until milestone 4. The member lists depend on what
// DCE keeps, so this program uses a fixed set of Array members.
import cs.system.reflection.BindingFlags;
class P {
	static function typeName(t:cs.system.Type):String return t == null ? "null" : t.ToString();
	static function sorted(a:Array<String>) { a.sort((x, y) -> x < y ? -1 : x > y ? 1 : 0); return a.join(","); }
	static function main() {
		var a = [3, 1, 2];
		a.push(4);
		Sys.println(a.join("-") + " " + a.indexOf(2));
		Sys.println("fields Array: " + Type.getInstanceFields(Array));
		Sys.println("fields resolved Array: " + Type.getInstanceFields(Type.resolveClass("Array")));
		Sys.println("instFields Array: " + sorted(Type.getInstanceFields(Array)));
		var t = cs.system.Type._GetType("Array");
		var flags = new cs.Flags(BindingFlags.Public) | BindingFlags.Instance | BindingFlags.Static | BindingFlags.DeclaredOnly;
		var ms = t.GetMethods(flags);
		Sys.println("Array companion: " + typeName(t) + (t.IsInterface ? " interface" : ""));
		for (i in 0...ms.Length) Sys.println("  " + Std.string(ms[i]));
	}
}

# Std notes A4: Thread.create needs the ThreadStart delegate conversion (M4). The second thread now
# releases a Lock that main waits on, instead of main sleeping 0.2 s, so its line cannot be late.
--- thread-create @gencs
import sys.thread.Thread;
import sys.thread.Mutex;
import sys.thread.Lock;
import sys.thread.Deque;

class P {
	static function main() {
		var cur = Thread.current();
		Sys.println(cur != null);
		var lock = new Lock();
		var res = new Deque<Int>();
		var th = Thread.create(() -> {
			res.add(42);
			lock.release();
		});
		Sys.println(lock.wait(5));
		Sys.println(res.pop(false));
		var ms = haxe.Timer.stamp();
		Sys.println(ms > 0);
		Sys.println(Thread.current() == cur);
		var lock2 = new Lock();
		var t2 = Thread.create(() -> { Sys.println(Thread.current() != cur); lock2.release(); });
		lock2.wait();
		var done = false;
		haxe.Timer.delay(() -> { done = true; Sys.println("timer"); }, 20);
		Sys.sleep(0.2);
		Sys.println(done);
	}
}

# Recorded #8314 divergence (casts notes §6.15), set aside: starting a thread needs a delegate conversion
# (sys.thread.Thread.create, or cs.system.threading.Thread with a ThreadStart), which hxc refuses as untyped
# code until M4. gencs: `round 0: 0 of 7 same as the first`, `done`. Expected [run]: casts §6.15's stated hxc
# output (one shared Array<Int> in every round, so every round prints).
--- cast-array-threads
// Several threads read one unserialized array through a typed field at once. gencs: every read is its own
// copy (0 of 7 shared). Recorded #8314 divergence: hxc returns one Array<Int> to all of them (7 of 7).
typedef TI = {val:Array<Int>};

class P {
	static function main() {
		for (round in 0...20) {
			var t:TI = haxe.Json.parse('{"val":[1,2,3]}');
			var results:Array<Dynamic> = [for (i in 0...8) null];
			var done = new sys.thread.Lock();
			var go = new sys.thread.Lock();
			for (i in 0...8)
				sys.thread.Thread.create(() -> {
					go.wait();
					var v = t.val;
					results[i] = v;
					done.release();
				});
			for (i in 0...8)
				go.release();
			for (i in 0...8)
				done.wait();
			var same = 0;
			for (i in 1...8)
				if (results[i] == results[0]) same++;
			if (round == 0 || same != 0) Sys.println("round " + round + ": " + same + " of 7 same as the first");
		}
		Sys.println("done");
	}
}

# Set aside from plan 3c Task 11: sys.Http.customRequest parses its URL with EReg.matched (std notes A5,
# M4), so the case throws NotSupportedException before it reads anything. http-response-reading covers
# readHttpResponse and getResponseHeaderValues in 3c.
# Std notes A6 (a fake socket, no network). getResponseHeaderValues is called only where a header
# repeats; gencs's NullReferenceException otherwise is kept as a gencs bug and not exercised. The header
# names are sorted with a comparator that returns 0 for equal strings: with the researcher's `a < b ? -1 : 1`,
# hxcs's quicksort gave an order that followed the StringMap's per-process key order (seen in repeated runs).
--- http-response-parsing @gencs
import haxe.io.Bytes;
import haxe.io.BytesInput;
import haxe.io.BytesOutput;

class Fake extends sys.net.Socket {
	public var sent:BytesOutput = new BytesOutput();
	public function new(response:String) {
		super();
		@:privateAccess this.input = new BytesInput(Bytes.ofString(response));
		@:privateAccess this.output = sent;
	}
	override public function connect(host:sys.net.Host, port:Int):Void {}
	override public function setTimeout(t:Float):Void {}
	override public function shutdown(r:Bool, w:Bool):Void {}
	override public function close():Void {}
}

class P {
	static function run(name:String, response:String, ?post:Bool = false) {
		var h = new sys.Http("http://127.0.0.1:8080/path?x=1");
		h.addHeader("X-A", "1");
		var out = new BytesOutput();
		var status = -1;
		h.onStatus = s -> status = s;
		h.onError = e -> Sys.println(name + " error: " + e);
		h.onData = d -> Sys.println(name + " data: " + StringTools.replace(d, "\n", "\n"));
		var f = new Fake(response);
		h.customRequest(post, out, f);
		Sys.println(name + " status " + status);
		if (h.responseHeaders == null) { Sys.println(name + " no headers"); return; }
		var keys = [for (k in h.responseHeaders.keys()) k];
		keys.sort((a, b) -> a < b ? -1 : a > b ? 1 : 0);
		Sys.println(name + " headers " + keys.map(k -> k + "=" + h.responseHeaders.get(k)).join(";"));
		if (name == "len") Sys.println(name + " multi " + h.getResponseHeaderValues("Set-Cookie"));
		Sys.println(name + " body " + StringTools.replace(out.getBytes().toString(), "\n", "\n"));
		Sys.println(name + " request " + StringTools.replace(StringTools.replace(f.sent.getBytes().toString(), "\r", "\r"), "\n", "\n"));
	}
	static function main() {
		run("len", "HTTP/1.1 200 OK\r\nContent-Length: 5\r\nX-B: b\r\nSet-Cookie: a=1\r\nSet-Cookie: b=2\r\n\r\nhello");
		run("eof", "HTTP/1.1 200 OK\r\nX-C: c\r\n\r\nuntil end");
		run("404", "HTTP/1.1 404 Not Found\r\nContent-Length: 2\r\n\r\nno");
		run("bad", "garbage\r\n\r\n");
		run("post", "HTTP/1.1 200 OK\r\nContent-Length: 0\r\n\r\n", true);
	}
}
