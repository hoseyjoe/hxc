/**
 * zlib (RFC 1950) compression for PDF FlateDecode streams. C# has no haxe.zip.Compress, so it uses
 * .NET's ZLibStream through the externs at the end of this file.
 */
class Zlib {
	public static function compress(b:Bytes):Bytes {
		var s = new ZlibStream();
		s.write(b, 0, b.length);
		return s.finish();
	}
}

/**
 * zlib compression fed piece by piece: on C# through .NET's ZLibStream as it comes, so only the compressed bytes are
 * held; elsewhere collected and compressed at the end.
 */
class ZlibStream {
	var ms:cs.system.io.MemoryStream;
	var z:NetZLibStream;

	public function new() {
		ms = new cs.system.io.MemoryStream();
		z = new NetZLibStream(ms, NetCompressionLevel.Optimal, true);
	}

	public function write(b:Bytes, pos:Int, len:Int):Void {
		z.Write(b.getData(), pos, len);
	}

	/** The compressed data; nothing more can be written. */
	public function finish():Bytes {
		z.Dispose();
		return Bytes.ofData(ms.ToArray());
	}
}

	function Dispose():Void;
}

	Optimal;
	Fastest;
	NoCompression;
	SmallestSize;
