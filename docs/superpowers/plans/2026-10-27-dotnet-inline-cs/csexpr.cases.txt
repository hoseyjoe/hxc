# A bare `throw` in inline C# is IL `rethrow`, which keeps the stack trace: in the catch, after a try/catch
# nested in it (Roslyn's `rethrow` inside a try within the handler verifies), after an if, in a loop, and in
# the cs.system.Exception catch the exceptions filter makes of a haxe.Exception catch (haxeui-core's
# ComponentBase/ValidationManager shape). gencs writes `throw;` for each.
--- inline-cs-rethrow-bare @gencs
// A bare `throw` in inline C# (untyped __cs__ or cs.Syntax.code) rethrows the caught exception and keeps its
// stack trace: directly in the catch, after a try/catch nested in it, after an if, in a loop, and in the catch
// a haxe.Exception catch becomes. The trace still names the method that threw.
class P {
	static function thrower():Void {
		throw new cs.system.InvalidOperationException("boom");
	}
	static function hxThrower():Void {
		throw new haxe.Exception("hx");
	}
	static function report(tag:String, e:cs.system.Exception) {
		Sys.println(tag + ": " + cs.Lib.getNativeType(e).Name + ", thrower in trace " + (e.StackTrace.indexOf("hrower") >= 0));
	}
	static function direct() {
		try thrower() catch (e:cs.system.Exception) untyped __cs__("throw");
	}
	static function syntax() {
		try thrower() catch (e:cs.system.Exception) cs.Syntax.code("throw");
	}
	static function afterNestedTry() {
		try thrower() catch (e:cs.system.Exception) {
			try throw "inner" catch (e2:Dynamic) {}
			untyped __cs__("throw");
		}
	}
	static function afterIf(n:Int) {
		var log = [];
		try thrower() catch (e:cs.system.Exception) {
			if (n > 0) log.push("cleanup");
			untyped __cs__("throw");
		}
	}
	static function inLoop(n:Int) {
		try thrower() catch (e:cs.system.Exception) {
			var i = 0;
			while (i < n) {
				if (i == 1) untyped __cs__("throw");
				i++;
			}
		}
	}
	static function haxeCatch() {
		try hxThrower() catch (e:haxe.Exception) {
			Sys.println("caught " + e.message);
			cs.Syntax.code("throw");
		}
	}
	static function main() {
		try direct() catch (e:cs.system.Exception) report("direct", e);
		try syntax() catch (e:cs.system.Exception) report("syntax", e);
		try afterNestedTry() catch (e:cs.system.Exception) report("after nested try", e);
		try afterIf(1) catch (e:cs.system.Exception) report("after if", e);
		try inLoop(3) catch (e:cs.system.Exception) report("in loop", e);
		try haxeCatch() catch (e:cs.system.Exception) report("haxe.Exception catch", e);
	}
}

# The spec's first-class rethrow: `throw e` of an unchanged catch variable, directly in its own catch, is
# `rethrow` in hxc. gencs writes `throw e;`, which resets the trace (the [gencs] section). A reassigned e, a
# `throw e` inside a nested catch (rethrow would throw e2) and the haxe.Exception and Dynamic catches (the
# filter throws Exception.thrown(e), a call's result) stay plain throws, as in gencs.
--- rethrow-throw-caught-value @gencs
// `throw e` of the caught value, unchanged, directly in its catch: hxc emits `rethrow`, which keeps the stack
// trace (the spec's first-class rethrow); gencs writes `throw e;`, which resets it. Reassigned, thrown from a
// nested catch, or caught as haxe.Exception or Dynamic (the filter wraps those), it stays a plain throw.
class MyErr extends haxe.Exception {}
class P {
	static function thrower():Void {
		throw new cs.system.InvalidOperationException("boom");
	}
	static function hxThrower():Void {
		throw new MyErr("hx");
	}
	static function report(tag:String, e:cs.system.Exception) {
		Sys.println(tag + ": " + cs.Lib.getNativeType(e).Name + ", thrower in trace " + (e.StackTrace.indexOf("hrower") >= 0));
	}
	static function native() {
		try thrower() catch (e:cs.system.Exception) { Sys.println("cleanup"); throw e; }
	}
	static function haxeSubclass() {
		try hxThrower() catch (e:MyErr) throw e;
	}
	static function reassigned() {
		try thrower() catch (e:cs.system.Exception) { e = new cs.system.Exception("other"); throw e; }
	}
	static function fromNestedCatch() {
		try thrower() catch (e:cs.system.Exception) {
			try throw "inner" catch (e2:Dynamic) throw e;
		}
	}
	static function haxeException() {
		try thrower() catch (e:haxe.Exception) throw e;
	}
	static function dynamicCatch() {
		try thrower() catch (e:Dynamic) throw e;
	}
	static function main() {
		try native() catch (e:cs.system.Exception) report("native", e);
		try haxeSubclass() catch (e:cs.system.Exception) report("haxe subclass", e);
		try reassigned() catch (e:cs.system.Exception) report("reassigned", e);
		try fromNestedCatch() catch (e:cs.system.Exception) report("from nested catch", e);
		try haxeException() catch (e:cs.system.Exception) report("haxe.Exception", e);
		try dynamicCatch() catch (e:cs.system.Exception) report("Dynamic", e);
	}
}

# Placeholder substitution and result conversion as gencs does them (NOTES section 2).
--- inline-cs-placeholders @gencs
// cs.Syntax.code and untyped __cs__ placeholders: each {N} is its argument's value as a unit (`(i + 1) * 3`),
// in the argument's own type (an Int stays an int, so `{0} * 2` is int arithmetic). A placeholder used twice
// evaluates its argument twice, as gencs's text substitution does. cs.Syntax.code's result is Dynamic,
// converted to where it goes (Runtime.toInt, toDouble, toString, toBool); untyped __cs__'s takes the type
// it is used as, with no conversion.
class P {
	var fld:Int = 7;
	static var sfld:String = "s";
	public function new() {}
	static function take(i:Int):Int return i * 2;
	static function side():Int { Sys.println("side"); return 3; }
	function inst():Void {
		var a:Int = untyped __cs__("{0} + 1", fld);
		var b:Int = cs.Syntax.code("{0} + 1", this.fld);
		Sys.println(a + " " + b);
		var c:String = cs.Syntax.code("{0}.ToUpper()", sfld);
		Sys.println(c);
	}
	static function main() {
		var i = 5;
		var f = 2.5;
		var s = "hi";
		var d:Dynamic = 4;
		var r1:Int = cs.Syntax.code("{0} * 2", i);
		var r2:Float = cs.Syntax.code("{0} * 2", f);
		var r3:String = cs.Syntax.code("{0} + \"!\"", s);
		var r4:Int = cs.Syntax.code("(int){0} + 1", d);
		var r7:Int = cs.Syntax.code("{0} + {0}", side());
		var r8 = take(cs.Syntax.code("{0}", 21));
		var r9:Int = cs.Syntax.code("{0} * 3", i + 1);
		var r10 = cs.Syntax.code("{0}", "lit");
		var r11:Int = cs.Syntax.code("{1} - {0}", 1, 10);
		Sys.println(r1 + "," + r2 + "," + r3 + "," + r4 + "," + r7 + "," + r8 + "," + r9 + "," + r10 + "," + r11);
		if (cs.Syntax.code("{0} > 3", i)) Sys.println("gt");
		var u:Int = untyped __cs__("i + 10");
		Sys.println(u);
		new P().inst();
	}
}

# Names that resolve the same in gencs's C# and in the Haxe scope.
--- inline-cs-names @gencs
// Names in inline C# that gencs's C# resolves as the Haxe code means them: locals, `this`, the class's own
// fields and statics unqualified, inherited fields, a static through its class's name, a method call, a local
// read in nested loops (the server app's `cells[row, col]` shape), and a private field of another class
// reached through a field (haxe-strings' `this.sb.b`).
class Base {
	public var inherited:Int = 40;
	public function new() {}
}
class Buf {
	var sb = new StringBuf();
	public function new() {}
	public function add(s:String):Buf { sb.add(s); return this; }
	public function clear():Buf { untyped __cs__("this.sb.b.Clear()"); return this; }
	public function insert(pos:Int, item:String):Buf { untyped __cs__("this.sb.b.Insert(pos, item)"); return this; }
	public function insertChar(pos:Int, ch:Int):Buf { untyped __cs__("this.sb.b.Insert(pos, (System.Char)ch)"); return this; }
	public function toString() return sb.toString();
}
class P extends Base {
	var fld:Int = 7;
	static var sfld:Int = 8;
	static function hello(x:Int) return "hello" + x;
	function inst():Void {
		Sys.println(untyped __cs__("fld + sfld + inherited"));
		Sys.println(untyped __cs__("this.fld + P.sfld"));
		Sys.println(untyped __cs__("hello(fld)"));
	}
	static function hashOf(str:String):Bool return untyped __cs__("str.GetHashCode() == {0}", (str : cs.system.String).GetHashCode());
	static function main() {
		var row = 0;
		var total = 0;
		var items = [1, 2, 3];
		for (item in items) {
			row++;
			var col = 0;
			while (col < item) {
				col++;
				total += untyped __cs__("row * 10 + col");
			}
		}
		Sys.println(total);
		new P().inst();
		var b = new Buf().add("held");
		b.insert(1, "XY").insertChar(0, 65);
		Sys.println(b.toString());
		Sys.println(b.clear().add("z").toString());
		Sys.println(hashOf("abc"));
	}
}

# The desktop app's and haxeui's forms beyond the spec's first list: `is`, keyword/array types, chained
# casts, comparisons and &&, int.MinValue, `new`, enum members, a property chain, a call on a cast, a call
# on a struct a method returns (needs a temporary and ldloca), and ZLibStream (a type no --net-std of the
# server app defines; hxc finds it through NetRefs, from the reference pack).
--- inline-cs-forms @gencs
// The other inline C# forms the desktop app and its libraries use: type tests (`is` with a dotted, keyword
// and array type), casts to keyword and array types and chained casts, comparisons and && on a long against
// int.MinValue/int.MaxValue with a placeholder used twice, `new` of a .NET class, enum members as arguments,
// a static property chain, a cast receiver with a chained call, and a call on a struct a method returns.
import haxe.io.Bytes;
class P {
	static function toHaxe(v:Dynamic):Dynamic {
		if (cs.Syntax.code("{0} is System.DBNull", v))
			return null;
		if (cs.Syntax.code("{0} is long", v)) {
			var l:cs.StdTypes.Int64 = cs.Syntax.code("(long){0}", v);
			if (cs.Syntax.code("({0} >= int.MinValue && {0} <= int.MaxValue)", l))
				return (cs.Syntax.code("(int){0}", l) : Int);
			return (cs.Syntax.code("(double){0}", l) : Float);
		}
		if (cs.Syntax.code("{0} is byte[]", v))
			return Bytes.ofData(cs.Syntax.code("(byte[]){0}", v));
		return v;
	}
	static var client:Dynamic = null;
	static function main() {
		var boxedLong:Dynamic = cs.system.Convert.ToInt64(5);
		var bigLong:Dynamic = cs.system.Convert.ToInt64(5000000000.0);
		var bytes:Dynamic = Bytes.ofString("ab").getData();
		Sys.println([toHaxe(cs.system.DBNull.Value), toHaxe(boxedLong), toHaxe(bigLong), (toHaxe(bytes) : Bytes).length, toHaxe("s")]);
		var id:Int = cs.Syntax.code("(int)(long){0}", boxedLong);
		Sys.println(id);
		Sys.println(cs.Syntax.code("global::System.Globalization.CultureInfo.InvariantCulture.Name") == "");
		var s = "Hello";
		Sys.println(cs.Syntax.code("{0}.StartsWith({1}, global::System.StringComparison.Ordinal)", s, "He"));
		Sys.println(cs.Syntax.code("{0}.EndsWith({1}, global::System.StringComparison.Ordinal)", "abc\u00AD", "abc"));
		client = cs.Syntax.code("new System.Text.StringBuilder()");
		var chained:String = cs.Syntax.code("((System.Text.StringBuilder){0}).Append({1}).ToString()", client, "chain");
		Sys.println(chained);
		var list = new cs.system.collections.generic.List_1<String>();
		list.Add("done");
		var moved:Bool = cs.Syntax.code("((System.Collections.Generic.List<string>){0}).GetEnumerator().MoveNext()", list);
		Sys.println(moved);
		var ms = new cs.system.io.MemoryStream();
		var z:cs.system.io.Stream = untyped __cs__("new global::System.IO.Compression.ZLibStream({0}, global::System.IO.Compression.CompressionLevel.Optimal, true)", ms);
		z.Write(Bytes.ofString("aaaaaaaaaa").getData(), 0, 10);
		z.Dispose();
		Sys.println(ms.ToArray().Length > 0);
	}
}

# `(decimal){0}` folds a constant as C# does (0 -> Decimal.Zero, 7 -> Decimal(int32), 2.5 -> the five-part
# constructor with scale 1) and calls op_Implicit(int32)/op_Explicit(float64) on a variable; `1.0m` keeps
# its scale. new Decimal(x) is the double constructor in both compilers (it already passes in hxc).
--- inline-cs-decimal @gencs
// `(decimal){0}` in inline C# (the server app's form) and decimal literals, against the Haxe forms that
// compile under both compilers: `new Decimal(x)` takes the double constructor (Haxe's overload choice), so a
// Float is rounded to 15 significant digits and 1.0 has scale 0, where the literal 1.0m has scale 1.
import cs.system.Decimal;
@:nativeGen
class Model {
	@:property

	public function new() {}
}

	static function scale(d:Decimal):Int return (Decimal.GetBits(d)[3] >> 16) & 0xFF;
	static function show(tag:String, d:Decimal) Sys.println(tag + " " + d + " scale " + scale(d));
	static function main() {

		show("(decimal)0", untyped __cs__("(decimal){0}", 0));

		show("(decimal)double", untyped __cs__("(decimal){0}", f));

	}

--- inline-cs-serverapp-patterns @gencs @netlib
// The server app's inline C# shapes, unchanged: a bare `throw` after cleanup in a catch of cs.system.Exception,
// `cells[row, col]` on locals in nested loops (ValLib.Cells stands in for EPPlus's ExcelRange), `(decimal){0}`

@:nativeGen
class CostBenefit {
	@:property
	public var value:Decimal = untyped __cs__("(decimal){0}", 0);
	public function new() {}
}
class Transaction {
	public function new() {}
	public function release() Sys.println("released");
}
class P {
	static function work(t:Transaction):Void {
		throw new cs.system.InvalidOperationException("failed");
	}
	static function handle() {
		var transaction:Transaction = null;
		try {
			transaction = new Transaction();
			work(transaction);
		} catch (e:cs.system.Exception) {
			if (transaction != null) transaction.release();
			untyped __cs__("throw"); // Throw the original exception to avoid lost stack trace
		}
	}
	static function sheet(rows:Array<Array<String>>):Int {
		var cells = new vallib.Cells();
		var sum = 0;
		var row = 0;
		for (r in rows) {
			row++;
			var col = 0;
			for (c in r) {
