@:native("System.ComponentModel.DataAnnotations.RangeAttribute") extern class RangeAttribute extends cs.system.Attribute {

@:nativeGen
@:strict(DescriptionAttribute("the model"))
class Search_Model {
	public function new() {}
	@:strict(DescriptionAttribute("Text to search for."))
	@:strict(RequiredAttribute)
	@:property public var text:String = "";
	@:strict(DescriptionAttribute("Maximum results."))
	@:property public var limit:Int = 10;
	@:strict(RequiredAttribute({AllowEmptyStrings: true}))
	@:strict(RangeAttribute(1, 5, {ErrorMessage: "out of range"}))
	public var plain:Int = 0;
	@:strict(RangeAttribute(1.5, 5))
	public var real:Float = 0;
	@:strict(DescriptionAttribute("a method")) public function run(@:strict(DescriptionAttribute("arg, not written")) q:String):Int return 0;
	@:strict(DescriptionAttribute("static")) public static function make():Search_Model return new Search_Model();
}

class P {
	static function show(a:Dynamic):String {
		if (Std.isOfType(a, DescriptionAttribute)) return "Description(" + (a : DescriptionAttribute).Description + ")";
		if (Std.isOfType(a, RangeAttribute)) { var r:RangeAttribute = a; return "Range(" + Std.string(r.Minimum) + " " + cs.Lib.toNativeType(Type.getClass(r.Minimum)).Name + ", " + r.ErrorMessage + ")"; }
		if (Std.isOfType(a, RequiredAttribute)) return "Required(" + (a : RequiredAttribute).AllowEmptyStrings + ")";
		return Std.string(a);
	}
	static function list(label:String, attrs:cs.NativeArray<Dynamic>) {
		Sys.println(label + ": " + [for (i in 0...attrs.Length) show(attrs[i])].join(" "));
	}
	static function main() {
		var t = cs.Lib.toNativeType(Search_Model);
		var all = new cs.Flags(BindingFlags.Public) | BindingFlags.Instance | BindingFlags.Static | BindingFlags.DeclaredOnly;

		for (n in ["plain", "real"]) list("field " + n, t.GetField(n, all).GetCustomAttributes(false));
		for (n in ["run", "make"]) list("method " + n, t.GetMethod(n, all).GetCustomAttributes(false));

		var bf = t.GetField("<text>k__BackingField", new cs.Flags(BindingFlags.NonPublic) | BindingFlags.Instance);
		list("backing field", bf.GetCustomAttributes(false));
	}
