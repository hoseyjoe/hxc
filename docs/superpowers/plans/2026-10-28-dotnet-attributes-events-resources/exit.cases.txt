# haxe.lang.Null<T> has the C# surface hxcs's @:classCode gives it: the implicit conversions from T and from
# __NoValue__ (C# code passing a bool to a Haxe ?x:Bool parameter needs the first: the server app's cs/Server
# and cs/SharedServer do it 14 times), the nested __NoValue__ class, _ofDynamic, and initonly fields.
# Today hxc prints "ofDynamic / False / false false / no op_Implicit(T)"; gencs's output is the expectation.
--- null-struct-csharp-surface @gencs
class P {
	static function main() {
		var t:cs.system.Type = cs.system.Type.GetType("haxe.lang.Null`1");
		var names = [];
		for (m in t.GetMethods()) {
			var n = m.Name;
			if (m.IsStatic && (n == "op_Implicit" || n == "_ofDynamic" || n == "ofDynamic")) names.push(n);
		}
		names.sort(Reflect.compare);
		Sys.println(names.join(","));
		Sys.println(t.GetNestedType("__NoValue__") != null);
		Sys.println(t.GetField("value").IsInitOnly + " " + t.GetField("hasValue").IsInitOnly);
		var ti = t.MakeGenericType(cs.NativeArray.make(cs.system.Type.GetType("System.Int32")));
		var conv = null;
		for (m in ti.GetMethods())
			if (m.Name == "op_Implicit" && m.GetParameters()[0].ParameterType == cs.system.Type.GetType("System.Int32")) conv = m;
		if (conv != null) Sys.println(conv.Invoke(null, cs.NativeArray.make((5 : Dynamic))));
		else Sys.println("no op_Implicit(T)");
	}
}

# A class whose only float-like reflection candidates are anonymous-structure fields gets gencs's own
# __hx_getField_f/__hx_setField_f overrides (gencs's is_dynamic counts a TAnon that isn't class/enum statics).
# Behaviour already agrees (both throw on the Float read and accept the Float write); only the declaring type
# of the override differs: hxc prints haxe.lang.HxObject on the first line today. Optional (reflection only).
--- reflect-getfield-f-anon-field @gencs
typedef Pt = {x:Int};
class C {
	public var p:Pt;
	public function new() {
		p = {x: 1};
	}
}
class P {
	static function main() {
		var t:cs.system.Type = cs.system.Type.GetType("C");
		var m = t.GetMethod("__hx_getField_f");
		Sys.println(m.DeclaringType.FullName);
		var d:Dynamic = new C();
		try {
			var g:Float = d.p;
			Sys.println(g);
		} catch (e:Dynamic) {
			Sys.println("threw " + Type.getClassName(Type.getClass(e)));
		}
		d.p = 3.5;
		Sys.println(d.p);
		var c:C = d;
		Sys.println(Std.string((cast c.p : Dynamic)));
		Sys.println(Reflect.field(d, "p"));
		Sys.println(Reflect.fields(d).join(","));
	}
}
