	Green;
	Blue;

enum E {
	A;
	B(x:Int, s:String);
	C(f:Float);
	D(e:E);
}
class P {
	static function name(c:Color):String {
		return switch (c) {
			case Red: "r";
			case Green: "g";
			case Blue: "b";
		}
	}
	static function desc(e:E):String {
		return switch (e) {
			case A: "a";
			case B(x, s) if (x > 10): "big " + s;
			case B(x, "q"): "q" + x;
			case B(1 | 2, s): "small " + s;
			case B(x, s): "b" + x + s;
			case D(D(inner)): "dd " + inner;
			case D(B(_, s)): "db " + s;
			case D(_): "d";
			case _: "other";
		}
	}
	static function main() {
		Sys.println(name(Red) + name(Green) + name(Blue));
		Sys.println(desc(A));
		Sys.println(desc(B(11, "x")));
		Sys.println(desc(B(1, "q")));

		Sys.println(e.match(B(4, _)));

			case null: "null";

		return switch (t) {
			case Leaf(v): v;
			case Node(l, r): sum(l) + sum(r);
		}
	}
	static function get<T>(o:Opt<T>, def:T):T {
		return switch (o) {
			case Some(v): v;
